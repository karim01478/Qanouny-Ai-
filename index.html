async function sendMessage() { const input = document.getElementById("userInput"); const chatBox = document.getElementById("chatBox"); const userMessage = input.value.trim(); if (!userMessage) return; // عرض رسالة المستخدم chatBox.innerHTML += `
${userMessage}
`; input.value = ""; // رسالة loading const loadingMsg = document.createElement("div"); loadingMsg.className = "bot-msg"; loadingMsg.innerText = "⏳ جاري المعالجة..."; chatBox.appendChild(loadingMsg); try { const response = await fetch("https://qanoun-ai-production.up.railway.app/chat", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ messages: [ { role: "user", content: userMessage } ] }) }); // تحقق من response if (!response.ok) { throw new Error("Server error: " + response.status); } const data = await response.json(); loadingMsg.remove(); console.log("API RESPONSE:", data); let reply = "⚠️ لا يوجد جواب"; // الحالة العادية if (data.choices && data.choices.length > 0) { reply = data.choices[0]?.message?.content || reply; } // إذا كان error من API if (data.error) { reply = "❌ خطأ: " + (data.error.message || data.error); } chatBox.innerHTML += `
${reply}
`; } catch (error) { loadingMsg.remove(); chatBox.innerHTML += `
❌ خطأ في الاتصال بالسيرفر
`; console.error("ERROR:", error); } chatBox.scrollTop = chatBox.scrollHeight; }
