async function sendMessage() { const input = document.getElementById("userInput"); const chatBox = document.getElementById("chatBox"); const userMessage = input.value.trim(); if (!userMessage) return; // عرض سؤال المستخدم chatBox.innerHTML += `
${userMessage}
`; input.value = ""; // رسالة loading const loadingMsg = document.createElement("div"); loadingMsg.className = "bot-msg"; loadingMsg.innerText = "⏳ جاري المعالجة..."; chatBox.appendChild(loadingMsg); try { const response = await fetch("https://qanoun-ai-production.up.railway.app/chat", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ model: "llama3-70b-8192", messages: [ { role: "system", content: "أنت مساعد قانوني مغربي. أجب بدقة وباللغة العربية." }, { role: "user", content: userMessage } ] }) }); const data = await response.json(); // مسح loading loadingMsg.remove(); let reply = "⚠️ لم يتم العثور على جواب"; if (data.choices && data.choices.length > 0) { reply = data.choices[0].message.content; } // عرض الجواب chatBox.innerHTML += `
${reply}
`; } catch (error) { loadingMsg.remove(); chatBox.innerHTML += `
❌ خطأ في الاتصال بالسيرفر
`; } // نزول تلقائي chatBox.scrollTop = chatBox.scrollHeight; }
